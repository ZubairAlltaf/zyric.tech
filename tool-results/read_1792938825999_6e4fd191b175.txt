   507→    <nav aria-label="Tool categories" className="flex h-full flex-col gap-5">

   549→/* ----------------------------- Command palette ------------------------------ */
   550→
   551→function CommandPalette({
   552→  open,
   553→  setOpen,
   554→  onOpenTool,
   555→}: {
   556→  open: boolean;
   557→  setOpen: (o: boolean) => void;
   558→  onOpenTool: (id: string) => void;
   559→}) {
   560→  const [q, setQ] = useState("");
   561→  const results = useMemo(() => searchTools(q), [q]);
   562→  const sig = results.filter((t) => t.category === "signature");
   563→  const rest = results.filter((t) => t.category !== "signature");
   564→
   565→  const pick = (id: string) => {
   566→    setOpen(false);
   567→    setQ("");
   568→    onOpenTool(id);
   569→  };
   570→
   571→  return (
   572→    <CommandDialog open={open} onOpenChange={setOpen} className="border-zinc-800 bg-zinc-950 text-zinc-100">
   573→      <CommandInput
   574→        placeholder={`Search ${TOOL_COUNT} tools…`}
   575→        value={q}
   576→        onValueChange={setQ}
   577→        className="text-zinc-100 placeholder:text-zinc-500"
   578→      />
   579→      <CommandList>
   580→        <CommandEmpty className="py-8 text-center text-sm text-zinc-500">
   581→          No tool matches “{q}”.
   582→        </CommandEmpty>
   583→        {sig.length > 0 && (
   584→          <CommandGroup heading="Signature tools">
   585→            {sig.map((t) => (
   586→              <CommandItem
   587→                key={t.id}
   588→                value={`${t.name} ${t.keywords}`}
   589→                onSelect={() => pick(t.id)}
   590→                className="gap-3 rounded-lg text-zinc-200 data-[selected=true]:bg-emerald-500/15 data-[selected=true]:text-emerald-300"
   591→              >
   592→                <t.icon className="h-4 w-4 text-amber-400" aria-hidden />
   593→                <span>{t.name}</span>
   594→                <span className="ml-auto text-[10px] uppercase tracking-wider text-amber-500/80">signature</span>
   595→              </CommandItem>
   596→            ))}
   597→          </CommandGroup>
   598→        )}
   599→        {rest.length > 0 && (
   600→          <CommandGroup heading="All tools">
   601→            {rest.map((t) => (
   602→              <CommandItem
   603→                key={t.id}
   604→                value={`${t.name} ${t.keywords}`}
   605→                onSelect={() => pick(t.id)}
   606→                className="gap-3 rounded-lg text-zinc-200 data-[selected=true]:bg-emerald-500/15 data-[selected=true]:text-emerald-300"
   607→              >
   608→                <t.icon className="h-4 w-4 text-emerald-400" aria-hidden />
   609→                <span>{t.name}</span>
   610→                <span className="ml-auto hidden text-[11px] text-zinc-600 sm:block">{t.description.slice(0, 40)}…</span>
   611→              </CommandItem>
   612→            ))}
   613→          </CommandGroup>
   614→        )}
   615→        <CommandSeparator />
   616→        <CommandGroup heading="Actions">
   617→          <CommandItem
   618→            value="surprise me random tool"
   619→            onSelect={() => pick(TOOLS[Math.floor(Math.random() * TOOLS.length)].id)}
   620→            className="gap-3 rounded-lg text-zinc-200 data-[selected=true]:bg-emerald-500/15 data-[selected=true]:text-emerald-300"
   621→          >
   622→            <Shuffle className="h-4 w-4 text-teal-400" aria-hidden /> Surprise me — open a random tool
   623→          </CommandItem>
   624→        </CommandGroup>
   625→      </CommandList>
   626→    </CommandDialog>
   627→  );
   628→}
